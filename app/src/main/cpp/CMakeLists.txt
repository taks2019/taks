cmake_minimum_required(VERSION 3.22.1)
project(TaksWorldNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(TAKS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../" ABSOLUTE)
add_library(taks_engine SHARED native_main.cpp FastScene.cpp RealModels.cpp "${TAKS_ROOT}/engine/CauseWorld.cpp" "${TAKS_ROOT}/engine/LivingWorld.cpp")
target_include_directories(taks_engine PRIVATE "${TAKS_ROOT}/engine")
find_library(gles_lib GLESv3)
find_library(log_lib log)
target_link_libraries(taks_engine PRIVATE ${gles_lib} ${log_lib})
