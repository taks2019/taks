cmake_minimum_required(VERSION 3.16)
project(TaksCauseEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(taks_cause STATIC CauseWorld.cpp LivingWorld.cpp)
target_include_directories(taks_cause PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
option(TAKS_BUILD_TESTS "Compile native engine tests" ON)
if(TAKS_BUILD_TESTS)
  enable_testing()
  add_executable(taks_engine_tests tests/WorldTests.cpp)
  target_link_libraries(taks_engine_tests PRIVATE taks_cause)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(taks_engine_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME taks_engine_causality COMMAND taks_engine_tests)
  add_executable(taks_living_world_tests tests/LivingWorldTests.cpp)
  target_link_libraries(taks_living_world_tests PRIVATE taks_cause)
  add_test(NAME taks_engine_ecology COMMAND taks_living_world_tests)
endif()
